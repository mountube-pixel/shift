1. src/js/store.js (Gestione Stato)

JavaScript

import { createStore } from 'framework7';
const store = createStore({
  state: {
    user: JSON.parse(localStorage.getItem('user')) || null,
    token: localStorage.getItem('token') || null,
    isLoading: false,
  },
  getters: {
    user({ state }) { return state.user; },
    isLoggedIn({ state }) { return !!state.user && !!state.token; },
    isLoading({ state }) { return state.isLoading; }
  },
  actions: {
    async login({ state }, { email, password }) {
      state.isLoading = true;
      try {
        const response = await fetch('https://shift.appap.it/api/v1/auth/login.php', {
          method: 'POST',
          body: JSON.stringify({ email, password }),
        });
        const data = await response.json();
        if (!response.ok) throw new Error(data.message);
        state.user = data.user;
        state.token = data.token;
        localStorage.setItem('user', JSON.stringify(data.user));
        localStorage.setItem('token', data.token);
        state.isLoading = false;
      } catch (error) {
        state.isLoading = false;
        throw error;
      }
    }
  },
});
export default store;

var routes = [
  {
    path: '/',
    component: HomePage, // Assumere HomePage esistente
    beforeEnter: function ({ resolve, reject }) {
      const router = this;
      if (store.getters.isLoggedIn.value) {
        resolve();
      } else {
        reject();
        router.navigate('/login/'); 
      }
    }
  },
  { path: '/login/', component: LoginPage },
  { path: '(.*)', component: NotFoundPage },
];
export default routes;
